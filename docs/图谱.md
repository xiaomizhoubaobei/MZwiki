# 词条关系图谱

:::tip 提示
- 🔵 **蓝色节点**：词条
- 🟠 **橙色节点**：标签
- 拖拽可移动节点，滚轮可缩放
- 点击词条节点可跳转到对应页面
:::

<div id="graph-container" style="width: 100%; height: 80vh; border: 1px solid var(--vp-c-divider); border-radius: 8px; overflow: hidden;"></div>

<script setup>
import { onMounted, ref } from 'vue'
import { data } from '../graph.data'

const container = ref(null)

onMounted(async () => {
  const ForceGraph = (await import('force-graph')).default

  const { nodes, links } = data

  ForceGraph()(container.value)
    .graphData({ nodes, links })
    .nodeLabel('id')
    .nodeColor((node) => (node.type === 'tag' ? '#ff9800' : '#42a5f5'))
    .nodeVal((node) => (node.type === 'tag' ? 6 : 10))
    .linkColor(() => 'rgba(120, 120, 120, 0.4)')
    .linkDirectionalArrowLength(4)
    .linkDirectionalArrowRelPos(1)
    .linkDirectionalParticles(1)
    .linkDirectionalParticleWidth(2)
    .onNodeClick((node) => {
      if (node.url) {
        const base = window.location.origin
        window.location.href = node.url.startsWith('/')
          ? base + node.url
          : base + '/' + node.url
      }
    })
    .backgroundColor('rgba(0, 0, 0, 0)')
})
</script>
