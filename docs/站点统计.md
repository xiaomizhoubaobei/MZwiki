<script setup>
import { useData } from 'vitepress'
import { computed, ref, onMounted } from 'vue'

const { theme } = useData()
const mounted = ref(false)
onMounted(() => { mounted.value = true })
const info = computed(() => theme.value.docAnalysisInfo || {})

const totalWords = computed(() => info.value.totalFileWords || 0)
const fileCount = computed(() => (info.value.fileList || []).length)
const lastCommit = computed(() => info.value.lastCommitTime || '未知')
const files = computed(() => (info.value.eachFileWords || []).filter(f => f && f.fileInfo))

function formatWords(count) {
  if (count < 1000) return count.toString()
  if (count < 10000) return (count / 1000).toFixed(1) + 'k'
  return (count / 10000).toFixed(1) + 'w'
}

// 按字数排序取前 20
const topFiles = computed(() => {
  const raw = (info.value.eachFileWords || [])
    .filter(f => f?.fileInfo?.relativePath && !f.fileInfo.relativePath.includes('index.md') && !f.fileInfo.relativePath.startsWith('标签/'))
    .map(f => ({
      title: (f.frontmatter && f.frontmatter.title) || f.fileInfo.relativePath,
      url: '/' + f.fileInfo.relativePath.replace(/\.md$/, ''),
      wordCount: f.wordCount || 0,
      readingTime: f.readingTime || '',
    }))
    .sort((a, b) => b.wordCount - a.wordCount)
    .slice(0, 20)
  return raw
})

const maxWordCount = computed(() => topFiles.value[0] ? topFiles.value[0].wordCount : 1)
</script>

<template>
  <div v-if="mounted" class="stats-page">
    <h1>📊 站点统计</h1>
    <p class="stats-desc">基于 vitepress-plugin-doc-analysis 自动生成的文档分析数据</p>

    <!-- 概览卡片 -->
    <div class="stats-overview">
      <div class="stat-card">
        <div class="stat-value">{{ fileCount }}</div>
        <div class="stat-label">📄 词条总数</div>
      </div>
      <div class="stat-card">
        <div class="stat-value">{{ formatWords(totalWords) }}</div>
        <div class="stat-label">📝 总字数</div>
      </div>
      <div class="stat-card">
        <div class="stat-value">{{ lastCommit }}</div>
        <div class="stat-label">🕐 最后更新</div>
      </div>
      <div class="stat-card">
        <div class="stat-value">{{ formatWords(Math.round(totalWords / 300)) }}</div>
        <div class="stat-label">⏱️ 总阅读时长(分钟)</div>
      </div>
    </div>

    <!-- 热门词条排行 -->
    <h2>🏆 热门词条（按字数排行）</h2>
    <div class="top-files">
      <div v-for="(file, index) in topFiles" :key="file.url" class="top-file-item">
        <span class="rank" :class="{ 'top-3': index < 3 }">{{ index + 1 }}</span>
        <div class="file-info">
          <a :href="file.url" class="file-title">
            {{ file.title }}
          </a>
          <span class="file-meta">
            {{ file.wordCount }} 字 · {{ file.readingTime }}
          </span>
        </div>
        <div class="word-bar">
          <div
            class="word-bar-fill"
            :style="{ width: (file.wordCount / maxWordCount * 100) + '%' }"
          ></div>
        </div>
      </div>
    </div>
  </div>
</template>
